it("persists Booking lifecycle with exact attribution, atomic validation, stable slots and complete pagination", async () => {
  const { t, client, orgId, recordB, snapshot } = await fixture();
  const actor = { kind: "user", id: (await client.query(api.users.me, {}))!._id };
  const company = await objectFields(client, orgId, "company");
  const companyRecord = await client.mutation(api.records.create, { orgId, objectId: company.object._id, values: { [company.fields.name._id]: "Local" } });
  const objectId = await client.mutation(api.objects.create, { orgId, key: "booking", label: "Booking", labelPlural: "Bookings" });
  const { fields } = await client.query(api.objects.get, { orgId, objectId });
  const name = fields.find(f => f.key === "name")!._id;
  const select = await client.mutation(api.fields.create, { orgId, objectId, key: "status", label: "Status", type: "select", options: [{ id: "new", label: "New" }, { id: "done", label: "Done" }] });
  const lookup = await client.mutation(api.fields.create, { orgId, objectId, key: "company", label: "Company", type: "lookup", targetObjectId: company.object._id });
  const values = { [name]: "Initial", [select.fieldId]: "new", [lookup.fieldId]: companyRecord.recordId };
  const created = await client.mutation(api.records.create, { orgId, objectId, values });
  expect((await client.query(api.records.get, { orgId, recordId: created.recordId }))!.record.values).toEqual(values);
  expect(await t.run(ctx => ctx.db.get(created.eventId))).toMatchObject({ actor, action: "create", before: null, after: values });
  const changes = { [name]: "Edited", [select.fieldId]: "done" };
  const updated = await client.mutation(api.records.update, { orgId, recordId: created.recordId, values: changes });
  expect(await t.run(ctx => ctx.db.get(updated.eventId))).toMatchObject({ actor, action: "update", before: { [name]: "Initial", [select.fieldId]: "new" }, after: changes });
  expect((await client.query(api.records.get, { orgId, recordId: created.recordId }))!.record.values).toEqual({ ...values, ...changes });
  for (const bad of [{ [select.fieldId]: "invalid" }, { [lookup.fieldId]: recordB.recordId }]) {
    const before = await snapshot();
    await expect(client.mutation(api.records.create, { orgId, objectId, values: { ...values, ...bad } })).rejects.toMatchObject({ data: { code: "VALIDATION" } });
    await expect(client.mutation(api.records.update, { orgId, recordId: created.recordId, values: { [name]: "Must roll back", ...bad } })).rejects.toMatchObject({ data: { code: "VALIDATION" } });
    expect(await snapshot()).toEqual(before);
  }
  const fieldBefore = await t.run(ctx => ctx.db.get(select.fieldId));
  await client.mutation(api.fields.update, { orgId, fieldId: select.fieldId, label: "Booking status" });
  expect(await t.run(ctx => ctx.db.get(select.fieldId))).toEqual({ ...fieldBefore, label: "Booking status" });
  const made = [];
  for (let i = 0; i < 9; i++) made.push(await client.mutation(api.fields.create, { orgId, objectId, key: `number${i}`, label: `Number ${i}`, type: "number" }));
  expect(made.map(f => f.slot)).toEqual([...Array.from({ length: 8 }, (_, index) => ({ kind: "n", index })), undefined]);
  await expect(client.query(api.records.list, { orgId, objectId, sort: { fieldId: made[8].fieldId, direction: "asc" }, paginationOpts })).rejects.toMatchObject({ data: { code: "UNINDEXED_FIELD" } });
  const removed = await client.mutation(api.records.remove, { orgId, recordId: created.recordId });
  expect(await client.query(api.records.get, { orgId, recordId: created.recordId })).toBeNull();
  expect(await t.run(ctx => ctx.db.get(removed.eventId))).toMatchObject({ actor, action: "delete", before: { ...values, ...changes }, after: null });
  expect((await client.query(api.events.forOrg, { orgId, paginationOpts: { numItems: 100, cursor: null } })).page.map(e => e._id)).toContain(removed.eventId);
  const expected = [];
  for (let i = 22; i >= 0; i--) {

    expected.unshift(row.recordId);
  }
  const ids: string[] = [], numbers: number[] = [], sizes: number[] = [];
  let cursor: string | null = null;

    sizes.push(result.page.length);

    if (result.isDone) break;
    expect(result.continueCursor).not.toBe(cursor); cursor = result.continueCursor;
